## 4. SectionRenderer Updates

Modify `artifacts/workshop/src/components/workshop/SectionRenderer.tsx`:

### Generic section support:
Add a new component `GenericSectionView`:

```tsx
function GenericSectionView({ section }: { section: GenericSectionData }) {
  return (
    <div className="animate-in fade-in slide-in-from-bottom-4 duration-500">
      <SectionHeader title={section.title} type={section.section_type} />
      {section.goal_text && <GoalBox text={section.goal_text} />}
      
      {/* Render rich text content as HTML */}
      <div className="prose max-w-none mb-8" dangerouslySetInnerHTML={{ __html: section.content }} />
      
      {/* Optional prompt block */}
      {section.prompt_block && (
        <div className="bg-primary rounded-lg p-6 text-white mb-8">
          <span className="inline-block bg-accent text-primary text-xs font-bold tracking-widest uppercase px-2.5 py-1 rounded mb-4">Prompt to Copy</span>
          <p className="text-white/90 text-sm leading-relaxed whitespace-pre-line">{section.prompt_block}</p>
          <CopyButton text={section.prompt_block} label="Copy Prompt" />
        </div>
      )}
      
      <NotesField sectionId={section.id} fieldKey="notes" label="Your Notes" />
    </div>
  );
}
```

### Update the switch statement default case:
```tsx
default:
  // Check if this is a generic section
  if (section.isGeneric && section.genericData) {
    return <GenericSectionView section={section.genericData} />;
  }
  return (
    <div className="p-10 border-2 border-dashed rounded-lg text-center">
      <h2 className="text-xl font-bold">{section.title}</h2>
      <p className="text-muted-foreground mt-2">Content coming soon...</p>
    </div>
  );
```

### Section data shape:
The `GET /sections` endpoint now returns generic section data inline for generic sections. The SectionRenderer needs to handle a section object that may include `isGeneric: true` and `genericData: { title, content, prompt_block, goal_text, section_type }`.

### Display name support:
The switch statement uses `section.id` to pick the component — that stays the same. But `SectionHeader` should use the section's `title` from the API response (which reflects the `display_name` override if one exists) rather than a hardcoded string. Update each section component to receive `title` as a prop instead of hardcoding it:

```tsx
// Before:
<SectionHeader title="Verification Test" type="exercise" />

// After:
<SectionHeader title={title || "Verification Test"} type="exercise" />
```

Pass `title` from the section data to each component via SectionRenderer.
